-- 面具系统核心管理器：初始面具选择、花纹生成与选择、增益叠加计算
-- 对应 C# Assets/Scripts/HotUpdate/XQL/Mask/MaskSystemManager.cs
--
-- ⚠️ 数据（allMaskDatas / allPatternDatas / randomBuffTypes）**保留在 C# 壳**，
--    因为那是 Inspector 里配好的序列化数据；Lua 通过壳上的下标接口读取。
--    本文件只维护「运行时状态」：已选花纹、总增益表、当前派系。

local behaviour = require 'behaviour'

local UnityEngine = CS.UnityEngine
local LuaBinding = CS.LuaBinding
local LuaUtil = CS.LuaUtil

local MaskSystemManager = {}

-- 对外返回的是「花纹在 allPatternDatas 里的下标」数组（用下标代替对象引用，
-- 跨语言传 List<PatternData> 很麻烦，传一组整数最稳）
function MaskSystemManager:start()
    local cs = self.cs

    self.totalBuffs = {}         -- { {buffType=, buffValue=, isPercent=}, ... }
    self.selectedPatterns = {}   -- 已选花纹下标
    self.currentMaskFaction = nil

    self:resolve_player_stats()
end

function MaskSystemManager:resolve_player_stats()
    local go = CS.SceneObjectManager.Instance:GetObjectByTag("Player")
    self.playerStats = (go ~= nil) and go:GetComponent(typeof(CS.PlayerStats)) or nil
end

-- ---------------- 增益表 ----------------

function MaskSystemManager:get_total_buff_value(buffType)
    local total = 0
    for i = 1, #self.totalBuffs do
        local buff = self.totalBuffs[i]
        if buff.buffType == buffType then
            total = total + buff.buffValue
        end
    end
    return total
end

local function add_buff(self, buffType, buffValue, isPercent)
    table.insert(self.totalBuffs, {
        buffType = buffType,
        buffValue = buffValue,
        isPercent = isPercent,
    })
end

-- 把某个初始面具的基础增益加进总表
function MaskSystemManager:add_mask_buffs(maskIndex)
    local cs = self.cs
    for j = 0, cs:GetMaskBaseBuffCount(maskIndex) - 1 do
        add_buff(self, cs:GetMaskBaseBuffType(maskIndex, j),
            cs:GetMaskBaseBuffValue(maskIndex, j), cs:GetMaskBaseBuffIsPercent(maskIndex, j))
    end
end

-- 把某个花纹的增益加进总表
function MaskSystemManager:add_pattern_buffs(patternIndex)
    local cs = self.cs
    for j = 0, cs:GetPatternBuffCount(patternIndex) - 1 do
        add_buff(self, cs:GetPatternBuffType(patternIndex, j),
            cs:GetPatternBuffValue(patternIndex, j), cs:GetPatternBuffIsPercent(patternIndex, j))
    end
end

-- 变化系：随机一条基础增益
function MaskSystemManager:add_random_base_buff()
    local cs = self.cs
    if cs:GetRandomBuffTypeCount() == 0 then
        UnityEngine.Debug.LogError("未配置变化系可随机的增益类型！")
        return
    end

    local idx = LuaUtil.RangeInt(0, cs:GetRandomBuffTypeCount())
    local buffType = cs:GetRandomBuffType(idx)
    local value = LuaUtil.Range(cs.randomBuffMinValue, cs.randomBuffMaxValue)
    add_buff(self, buffType, value, true)
    UnityEngine.Debug.Log(string.format("变化面具基础增益：%s +%.1f%%", tostring(buffType), value))
end

-- 变化系花纹：随机一条增益
function MaskSystemManager:add_random_pattern_buff()
    local cs = self.cs
    if cs:GetRandomBuffTypeCount() == 0 then
        UnityEngine.Debug.LogError("未配置变化系可随机的增益类型！")
        return
    end

    local idx = LuaUtil.RangeInt(0, cs:GetRandomBuffTypeCount())
    local buffType = cs:GetRandomBuffType(idx)
    local value = LuaUtil.Range(cs.randomBuffMinValue, cs.randomBuffMaxValue)
    add_buff(self, buffType, value, true)
    UnityEngine.Debug.Log(string.format("变化系花纹增益：%s +%.1f%%", tostring(buffType), value))
end

-- ---------------- 初始面具选择 ----------------

function MaskSystemManager:select_initial_mask(faction)
    local cs = self.cs

    if cs.IsMaskSelected then
        UnityEngine.Debug.LogWarning("已选择过初始面具，无法重复选择！")
        return
    end

    -- 查找对应面具数据
    local targetIndex = -1
    for i = 0, cs:GetMaskDataCount() - 1 do
        if cs:GetMaskDataFaction(i) == faction then targetIndex = i; break end
    end
    if targetIndex < 0 then
        UnityEngine.Debug.LogError("未找到对应的初始面具数据！")
        return
    end

    self.currentMaskFaction = faction
    cs.IsMaskSelected = true -- 单一真相源放在 C# 属性上（PlayerStats 会读它）

    if faction == CS.MaskFaction.Random then
        self:add_random_base_buff()
    else
        self:add_mask_buffs(targetIndex)
    end

    UnityEngine.Debug.Log(string.format("成功选择初始面具：%s，获得基础增益！", cs:GetMaskDataName(targetIndex)))
    self:on_mask_or_pattern_changed()
end

-- ---------------- 花纹：生成与选择 ----------------

-- 返回 3 个可选花纹的**下标**（UI 用），失败返回 nil
function MaskSystemManager:unlock_pattern_selection()
    local cs = self.cs

    if not cs.IsMaskSelected then
        UnityEngine.Debug.LogWarning("未选择初始面具，无法解锁花纹！")
        return nil
    end

    if #self.selectedPatterns >= cs.maxPatternCount then
        UnityEngine.Debug.LogWarning("已达到最大花纹数量，无法继续解锁！")
        return nil
    end

    return self:generate_optional_patterns(3)
end

function MaskSystemManager:generate_optional_patterns(count)
    local cs = self.cs
    local sameFaction, otherFaction = {}, {}

    for i = 0, cs:GetPatternDataCount() - 1 do
        if cs:IsPatternValid(i) then
            if cs:GetPatternFaction(i) == self.currentMaskFaction then
                table.insert(sameFaction, i)
            else
                table.insert(otherFaction, i)
            end
        end
    end

    if #sameFaction == 0 and #otherFaction == 0 then
        UnityEngine.Debug.LogError("【面具系统】无可用的有效花纹数据！请在Inspector配置花纹！")
        return {}
    end

    -- 同派系为空时强制走其他派系
    local forceUseOther = #sameFaction == 0
    local result = {}

    for _ = 1, count do
        local useSame = (not forceUseOther) and LuaUtil.Value01() <= cs.sameFactionProbability

        local pick
        if useSame and #sameFaction > 0 then
            pick = sameFaction[LuaUtil.RangeInt(1, #sameFaction + 1)]
        elseif #otherFaction > 0 then
            pick = otherFaction[LuaUtil.RangeInt(1, #otherFaction + 1)]
        else
            pick = #sameFaction > 0 and sameFaction[1] or otherFaction[1]
        end

        table.insert(result, pick)
    end

    -- 兜底：保证数量等于 count
    while #result < count and #result > 0 do
        table.insert(result, result[1])
    end

    return result
end

-- 选择花纹（patternIndex 是 allPatternDatas 里的下标）
function MaskSystemManager:select_pattern(patternIndex)
    local cs = self.cs

    if not cs.IsMaskSelected then
        UnityEngine.Debug.LogWarning("未选择初始面具，无法选择花纹！")
        return
    end

    if #self.selectedPatterns >= cs.maxPatternCount then
        UnityEngine.Debug.LogWarning("已达到最大花纹数量，无法继续选择！")
        return
    end

    if patternIndex == nil or patternIndex < 0 then
        UnityEngine.Debug.LogError("选择的花纹数据为空！")
        return
    end

    table.insert(self.selectedPatterns, patternIndex)

    if cs:GetPatternIsRandomEffect(patternIndex) then
        self:add_random_pattern_buff()
    else
        self:add_pattern_buffs(patternIndex)
    end

    self:on_mask_or_pattern_changed()
end

-- ---------------- 状态变化回调 ----------------

function MaskSystemManager:on_mask_or_pattern_changed()
    local panel = CS.UIManager.Instance:getPanelByName("PlayerProPanel")
    if not LuaBinding.IsAlive(panel) then
        UnityEngine.Debug.Log("xxa")
        return
    end

    if not LuaBinding.IsAlive(self.playerStats) then
        self:resolve_player_stats()
        if not LuaBinding.IsAlive(self.playerStats) then return end
    end

    -- 原逻辑：血量上限内补 50 点血
    local stats = self.playerStats
    stats.CurrentHealth = math.min(panel.currentMaxHealth, stats.CurrentHealth + 50)
    UnityEngine.Debug.Log("当前血量：" .. tostring(stats.CurrentHealth))
end

-- ---------------- 对外查询 ----------------

function MaskSystemManager:get_current_pattern_count() return #self.selectedPatterns end
function MaskSystemManager:get_current_mask_faction() return self.currentMaskFaction end

return function(cs)
    return behaviour.new(cs, MaskSystemManager)
end
