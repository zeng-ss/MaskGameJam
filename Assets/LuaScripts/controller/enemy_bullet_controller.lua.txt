-- 敌人子弹：直线飞行 5 秒后回收，命中玩家扣血
-- 对应 C# Assets/Scripts/HotUpdate/Controller/EnemyBulletController.cs

local behaviour = require 'behaviour'

local UnityEngine = CS.UnityEngine
local Vector3 = UnityEngine.Vector3
local Space = UnityEngine.Space
local Time = UnityEngine.Time
local PoolMgr = CS.PoolMgr
local SoundAudioPool = CS.SoundAudioPool
local LuaBinding = CS.LuaBinding

local EnemyBulletController = {}

local BULLET_SPEED = 20
local LIFETIME = 5

function EnemyBulletController:update()
    -- transform.Translate(Vector3.forward * (20 * dt), Space.Self)
    self.cs.transform:Translate(Vector3.forward * (BULLET_SPEED * Time.deltaTime), Space.Self)

    self.timer = (self.timer or 0) + Time.deltaTime
    if self.timer >= LIFETIME then
        PoolMgr.Instance:PushObj(self.bulletType, self.cs.gameObject)
        self.timer = 0
    end
end

-- 由 RobotController 调用（C# 壳 Init 转发过来，或 Lua 侧直接调）
function EnemyBulletController:init(bulletType, hit)
    self.bulletType = bulletType
    self.hitBloodValue = hit
end

function EnemyBulletController:ontriggerenter(other)
    if not other:CompareTag("Player") then return end

    local pool = SoundAudioPool.Instance
    pool:PlaySound(pool.hitPlayerClip, self.cs.transform.position)

    local player = other.gameObject:GetComponent(typeof(CS.PlayerStats))
    if LuaBinding.IsAlive(player) then
        local playerLua = LuaBinding.GetLuaInstance(player)
        if playerLua ~= nil then playerLua:take_damage(self.hitBloodValue or 0) end
    end
    PoolMgr.Instance:PushObj(self.bulletType, self.cs.gameObject)
end

return function(cs)
    return behaviour.new(cs, EnemyBulletController)
end
