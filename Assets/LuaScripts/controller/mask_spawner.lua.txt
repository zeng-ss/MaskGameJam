-- 面具护盾生成器：绕原点均匀生成 3 个面具
-- 对应 C# Assets/Scripts/HotUpdate/Controller/MaskSpawner.cs

local behaviour = require 'behaviour'

local UnityEngine = CS.UnityEngine
local Mathf = UnityEngine.Mathf
local Vector3 = UnityEngine.Vector3
local ResMgr = CS.ResMgr
local LuaBinding = CS.LuaBinding

local MaskSpawner = {}

local MASK_COUNT = 3
local RADIUS = 40
local HEIGHT = 1

function MaskSpawner:start()
    local angleStep = 360 / MASK_COUNT
    for i = 0, MASK_COUNT - 1 do
        -- 注意：Lua 的 for 循环变量每次迭代都是新的局部变量，
        -- 所以下面闭包捕获的 i 是正确的（不像 C# 的 for 循环需要额外拷贝）
        ResMgr.Instance:LoadAndInstantiateAsync("Mask", nil, function(mask)
            if not LuaBinding.IsAlive(mask) then
                UnityEngine.Debug.LogError("护盾预制体加载失败！")
                return
            end
            mask.name = "Mask_" .. i
            -- 角度转弧度（用 math.rad，Mathf.Deg2Rad 是 const，Lua 侧取不到）
            local currentAngle = math.rad(angleStep * i)
            local x = Mathf.Cos(currentAngle) * RADIUS
            local z = Mathf.Sin(currentAngle) * RADIUS
            mask.transform.position = Vector3(x, HEIGHT, z)
        end)
    end
end

return function(cs)
    return behaviour.new(cs, MaskSpawner)
end
